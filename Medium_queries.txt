-- Q101. Display customers along with their plan name.
SELECT c.id, c.name, p.name AS plan_name
FROM customers c
JOIN plans p ON c.plan_id = p.id;


-- Q102. Display customer name, connection ID and plan price.
SELECT c.name, c.connection_id, p.price
FROM customers c
JOIN plans p ON c.plan_id = p.id;


-- Q103. Display all transactions with customer names.
SELECT t.id, c.name AS customer_name, t.amount, t.payment_mode, t.status
FROM transactions t
JOIN customers c ON t.customer_id = c.id;


-- Q104. Display usage logs with customer names.
SELECT u.id, c.name AS customer_name, u.usage_date, u.total_usage_gb
FROM usage_logs u
JOIN customers c ON u.customer_id = c.id;


-- Q105. Display customers and their plan speed.
SELECT c.name, p.name AS plan_name, p.speed
FROM customers c
JOIN plans p ON c.plan_id = p.id;


-- Q106. Display customer name, plan name and validity days.
SELECT c.name, p.name AS plan_name, p.validity_days
FROM customers c
JOIN plans p ON c.plan_id = p.id;


-- Q107. Display successful transactions with customer names.
SELECT t.id, c.name AS customer_name, t.amount, t.date
FROM transactions t
JOIN customers c ON t.customer_id = c.id
WHERE t.status = 'Success';


-- Q108. Display failed transactions with customer names.
SELECT t.id, c.name AS customer_name, t.amount, t.date
FROM transactions t
JOIN customers c ON t.customer_id = c.id
WHERE t.status = 'Failed';


-- Q109. Display customers whose plan costs more than 1000.
SELECT c.name, p.name AS plan_name, p.price
FROM customers c
JOIN plans p ON c.plan_id = p.id
WHERE p.price > 1000;


-- Q110. Display customers using plans with speed greater than 100 Mbps.
SELECT c.name, p.name AS plan_name, p.speed
FROM customers c
JOIN plans p ON c.plan_id = p.id

-- Q111. Display users along with their customer names.
SELECT u.username, u.role, c.name AS customer_name
FROM users u
LEFT JOIN customers c ON u.customer_id = c.id;


-- Q112. Display password reset records with usernames.
SELECT pr.id, u.username, pr.otp, pr.created_at, pr.expires_at
FROM password_resets pr
JOIN users u ON pr.user_id = u.id;


-- Q113. Display renewal alerts along with customer names.
SELECT r.id, c.name AS customer_name, r.alert_date
FROM renewal_alert_logs r
JOIN customers c ON r.customer_id = c.id;


-- Q114. Display transaction details with customer and plan information.
SELECT t.id,
       c.name AS customer_name,
       p.name AS plan_name,
       t.amount,
       t.status
FROM transactions t
JOIN customers c ON t.customer_id = c.id
JOIN plans p ON c.plan_id = p.id;


-- Q115. Display usage details with customer and plan information.
SELECT u.id,
       c.name AS customer_name,

       u.usage_date,
       u.total_usage_gb
FROM usage_logs u
JOIN customers c ON u.customer_id = c.id
JOIN plans p ON c.plan_id = p.id;

FROM plans p

-- Q156. Categorize plans based on price.
SELECT name, price,
       CASE
           WHEN price < 500 THEN 'Budget'
           WHEN price BETWEEN 500 AND 1000 THEN 'Standard'
           ELSE 'Premium'
       END AS plan_category
FROM plans;


-- Q157. Categorize transactions based on amount.
SELECT id, amount,
       CASE
           WHEN amount < 500 THEN 'Low'
           WHEN amount BETWEEN 500 AND 1000 THEN 'Medium'
           ELSE 'High'
       END AS transaction_category
FROM transactions;


-- Q158. Display transaction status as Success/Not Successful.
SELECT id, amount,
       CASE
           WHEN status = 'Success' THEN 'Successful'
           ELSE 'Not Successful'
       END AS payment_result
FROM transactions;


-- Q159. Categorize customers based on plan price.

       CASE
           WHEN p.price < 500 THEN 'Budget'
           WHEN p.price <= 1000 THEN 'Standard'
           ELSE 'Premium'
       END AS category
FROM customers c
JOIN plans p ON c.plan_id = p.id;


-- Q160. Categorize usage records based on data usage.

       CASE
           WHEN total_usage_gb < 20 THEN 'Low Usage'
           WHEN total_usage_gb <= 50 THEN 'Medium Usage'
           ELSE 'High Usage'
       END AS usage_category
FROM usage_logs;

-- Q196. Find total successful payments made by customers of each plan.
SELECT p.name AS plan_name,
       SUM(t.amount) AS total_successful_payment
FROM plans p
JOIN customers c ON p.id = c.plan_id
JOIN transactions t ON c.id = t.customer_id
WHERE t.status = 'Success'
GROUP BY p.id, p.name;


-- Q197. Find average usage for each plan.
SELECT p.name AS plan_name,
       AVG(u.total_usage_gb) AS average_usage
FROM plans p
JOIN customers c ON p.id = c.plan_id
JOIN usage_logs u ON c.id = u.customer_id
GROUP BY p.id, p.name;


-- Q198. Find the number of customers and total transactions for each plan.
SELECT p.name AS plan_name,
       COUNT(DISTINCT c.id) AS customer_count,
       COUNT(t.id) AS transaction_count
FROM plans p
LEFT JOIN customers c ON p.id = c.plan_id
LEFT JOIN transactions t ON c.id = t.customer_id
GROUP BY p.id, p.name;


-- Q199. Find customers whose total usage is greater than their plan data limit.
-- This assumes data_limit_gb stores the numeric limit.
SELECT c.name,
       p.name AS plan_name,
       p.data_limit_gb,
       SUM(u.total_usage_gb) AS total_usage
FROM customers c
JOIN plans p ON c.plan_id = p.id
JOIN usage_logs u ON c.id = u.customer_id
WHERE p.data_limit_gb IS NOT NULL
GROUP BY c.id, c.name, p.id, p.name, p.data_limit_gb
HAVING SUM(u.total_usage_gb) > p.data_limit_gb;


-- Q200. Find customers whose total successful payments are greater than their plan price.
SELECT c.name,
       p.name AS plan_name,
       p.price,
       SUM(t.amount) AS total_paid
FROM customers c
JOIN plans p ON c.plan_id = p.id
JOIN transactions t ON c.id = t.customer_id
WHERE t.status = 'Success'
GROUP BY c.id, c.name, p.id, p.name, p.price
HAVING SUM(t.amount) > p.price;
