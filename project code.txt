-- 2. CREATE PLANS TABLE
-- ============================================================

CREATE TABLE plans (
    id INT PRIMARY KEY AUTO_INCREMENT,
    name VARCHAR(100) NOT NULL,
    speed VARCHAR(50) NOT NULL,
    data_limit VARCHAR(50) NOT NULL,
    data_limit_gb DECIMAL(10,2),
    validity_days INT NOT NULL,
    price DECIMAL(10,2) NOT NULL
);

-- ============================================================
-- 4. CREATE CUSTOMERS TABLE
-- ============================================================

CREATE TABLE customers (
    id INT PRIMARY KEY AUTO_INCREMENT,
    name VARCHAR(100) NOT NULL,
    connection_id VARCHAR(50) UNIQUE NOT NULL,
    email VARCHAR(150) UNIQUE NOT NULL,
    address VARCHAR(255),
    plan_id INT,
    start_date DATE,
    due_date DATE,
    followed_up BOOLEAN DEFAULT FALSE,

    FOREIGN KEY (plan_id)
        REFERENCES plans(id)
);

-- ============================================================
-- 10. CREATE TRANSACTIONS TABLE
-- ============================================================

CREATE TABLE transactions (
    id INT PRIMARY KEY AUTO_INCREMENT,
    customer_id INT NOT NULL,
    plan_id INT NOT NULL,
    amount DECIMAL(10,2) NOT NULL,
    payment_mode VARCHAR(50) NOT NULL,
    date DATE NOT NULL,
    status VARCHAR(20) NOT NULL,

    FOREIGN KEY (customer_id)
        REFERENCES customers(id),

    FOREIGN KEY (plan_id)
        REFERENCES plans(id)
);

        ROW_NUMBER() OVER (
            PARTITION BY customer_id
            ORDER BY date ASC, id ASC
        ) AS first_rn,
        ROW_NUMBER() OVER (
            PARTITION BY customer_id
            ORDER BY date DESC, id DESC
        ) AS last_rn
    FROM transactions
),
comparison AS (
    SELECT
        customer_id,
        MAX(CASE WHEN first_rn = 1 THEN amount END) AS first_amount,
        MAX(CASE WHEN last_rn = 1 THEN amount END) AS latest_amount
    FROM ranked
    GROUP BY customer_id
)

    customer_id,
    first_amount,
    latest_amount
FROM comparison
WHERE latest_amount > first_amount;

-- =========================================================
-- 295. Month-over-month revenue change
-- =========================================================
WITH monthly_revenue AS (
    SELECT
        DATE_FORMAT(date, '%Y-%m') AS month,
        SUM(amount) AS revenue
    FROM transactions
    GROUP BY DATE_FORMAT(date, '%Y-%m')
),
comparison AS (
    SELECT
        month,
        revenue,
        LAG(revenue) OVER (
            ORDER BY month
        ) AS previous_month_revenue
    FROM monthly_revenue
)
SELECT
    month,
    revenue,
    previous_month_revenue,
    revenue - previous_month_revenue AS revenue_change,
    ROUND(
        (revenue - previous_month_revenue) * 100.0 /
        NULLIF(previous_month_revenue, 0),
        2
    ) AS growth_percentage
FROM comparison;

    SELECT
        month,
        revenue,
        revenue - LAG(revenue) OVER (

    FROM monthly_revenue

    ON c.id = u.customer_id
ORDER BY c.id, u.date, u.id;


-- =========================================================
-- 299. Customers whose latest spending increased
-- compared with previous transaction
-- =========================================================
WITH ranked AS (
    SELECT
        t.*,
        LAG(amount) OVER (
            PARTITION BY customer_id
            ORDER BY date, id
        ) AS previous_amount,
        ROW_NUMBER() OVER (
            PARTITION BY customer_id
            ORDER BY date DESC, id DESC
        ) AS rn
    FROM transactions t
)
SELECT
    c.id,
    c.name,
    r.amount AS latest_amount,
    r.previous_amount,
    r.amount - r.previous_amount AS increase_amount
FROM customers c
JOIN ranked r
    ON c.id = r.customer_id
WHERE r.rn = 1
  AND r.previous_amount IS NOT NULL
  AND r.amount > r.previous_amount;


-- =========================================================
-- 300. Complete customer intelligence report
-- =========================================================
WITH transaction_summary AS (
    SELECT
        customer_id,
        COUNT(*) AS transaction_count,
        SUM(amount) AS total_spent,
        AVG(amount) AS average_transaction,
        MAX(date) AS last_transaction_date
    FROM transactions
    GROUP BY customer_id
),
usage_summary AS (
    SELECT
        customer_id,
        COUNT(*) AS usage_records,
        SUM(data_consumed) AS total_usage,
        MAX(date) AS last_usage_date
    FROM usage_logs
    GROUP BY customer_id
)
SELECT
    c.id AS customer_id,
    c.name AS customer_name,
    p.name AS plan_name,
    p.price AS plan_price,

    COALESCE(ts.transaction_count, 0) AS transaction_count,
    COALESCE(ts.total_spent, 0) AS total_spent,
    COALESCE(ts.average_transaction, 0) AS average_transaction,

    COALESCE(us.usage_records, 0) AS usage_records,
    COALESCE(us.total_usage, 0) AS total_usage,

    ts.last_transaction_date,
    us.last_usage_date,

    CASE
        WHEN ts.customer_id IS NULL
             AND us.customer_id IS NULL
            THEN 'Inactive'
        WHEN ts.customer_id IS NULL
            THEN 'Usage Only'
        WHEN us.customer_id IS NULL
            THEN 'Transaction Only'
        ELSE 'Active'
    END AS customer_status

FROM customers c
LEFT JOIN plans p
    ON c.plan_id = p.id
LEFT JOIN transaction_summary ts
    ON c.id = ts.customer_id
LEFT JOIN usage_summary us
    ON c.id = us.customer_id
ORDER BY total_spent DESC;
