-- 1. CREATE DATABASE
-- ============================================================

DROP DATABASE IF EXISTS broadband_db;

CREATE DATABASE broadband_db;

USE broadband_db;


-- ============================================================
-- 2. CREATE PLANS TABLE
-- ============================================================

CREATE TABLE plans (
    id INT PRIMARY KEY AUTO_INCREMENT,
    name VARCHAR(100) NOT NULL,
    speed INT NOT NULL,
    data_limit VARCHAR(50),
    data_limit_gb DECIMAL(10,2),
    validity_days INT NOT NULL,
    price DECIMAL(10,2) NOT NULL
);


-- ============================================================
-- 3. CREATE CUSTOMERS TABLE
-- ============================================================

CREATE TABLE customers (
    id INT PRIMARY KEY AUTO_INCREMENT,
    name VARCHAR(100) NOT NULL,
    connection_id VARCHAR(50) NOT NULL UNIQUE,
    email VARCHAR(100) NOT NULL UNIQUE,
    address VARCHAR(255),
    plan_id INT NOT NULL,
    start_date DATE NOT NULL,
    due_date DATE NOT NULL,
    followed_up BOOLEAN DEFAULT FALSE,

    CONSTRAINT fk_customer_plan
        FOREIGN KEY (plan_id)
        REFERENCES plans(id)
);


-- ============================================================
-- 4. CREATE USERS TABLE
-- ============================================================

CREATE TABLE users (
    id INT PRIMARY KEY AUTO_INCREMENT,
    username VARCHAR(100) NOT NULL UNIQUE,
    password_hash VARCHAR(255) NOT NULL,
    role ENUM('admin', 'staff', 'customer') NOT NULL,
    display_name VARCHAR(100),
    email VARCHAR(100) UNIQUE,
    customer_id INT NULL,

    CONSTRAINT fk_user_customer
        FOREIGN KEY (customer_id)
        REFERENCES customers(id)
);


-- ============================================================
-- 5. CREATE PASSWORD_RESETS TABLE
-- ============================================================

CREATE TABLE password_resets (
    id INT PRIMARY KEY AUTO_INCREMENT,
    user_id INT NOT NULL,
    otp VARCHAR(10) NOT NULL,
    created_at DATETIME NOT NULL,
    expires_at DATETIME NOT NULL,

    CONSTRAINT fk_password_reset_user
        FOREIGN KEY (user_id)
        REFERENCES users(id)
);


-- ============================================================
-- 6. CREATE USAGE_LOGS TABLE
-- ============================================================

CREATE TABLE usage_logs (
    id INT PRIMARY KEY AUTO_INCREMENT,
    customer_id INT NOT NULL,
    usage_date DATE NOT NULL,
    download_gb DECIMAL(10,2) DEFAULT 0,
    upload_gb DECIMAL(10,2) DEFAULT 0,
    total_usage_gb DECIMAL(10,2) DEFAULT 0,

    CONSTRAINT fk_usage_customer
        FOREIGN KEY (customer_id)
        REFERENCES customers(id)
);


-- ============================================================
-- 7. CREATE TRANSACTIONS TABLE
-- ============================================================

CREATE TABLE transactions (
    id INT PRIMARY KEY AUTO_INCREMENT,
    customer_id INT NOT NULL,
    amount DECIMAL(10,2) NOT NULL,
    payment_mode ENUM('Cash', 'UPI', 'Card', 'Net Banking') NOT NULL,
    status ENUM('Success', 'Pending', 'Failed') NOT NULL,
    date DATETIME NOT NULL,

    CONSTRAINT fk_transaction_customer
        FOREIGN KEY (customer_id)
        REFERENCES customers(id)
);


-- ============================================================
-- 8. CREATE RENEWAL_ALERT_LOGS TABLE
-- ============================================================

CREATE TABLE renewal_alert_logs (
    id INT PRIMARY KEY AUTO_INCREMENT,
    customer_id INT NOT NULL,
    alert_date DATETIME NOT NULL,
    alert_type VARCHAR(50) NOT NULL,
    status ENUM('Sent', 'Pending', 'Failed') NOT NULL,

    CONSTRAINT fk_alert_customer
        FOREIGN KEY (customer_id)
        REFERENCES customers(id)
);
