-- Q201. Find customers whose plan price is greater than the average plan price.
SELECT c.id, c.name, p.name AS plan_name, p.price
FROM customers c
JOIN plans p ON c.plan_id = p.id
WHERE p.price > (
    SELECT AVG(price)
    FROM plans
);


-- Q202. Find customers using the most expensive plan.
SELECT c.id, c.name, p.name AS plan_name, p.price
FROM customers c
JOIN plans p ON c.plan_id = p.id
WHERE p.price = (
    SELECT MAX(price)
    FROM plans
);


-- Q203. Find customers using the cheapest plan.
SELECT c.id, c.name, p.name AS plan_name, p.price
FROM customers c
JOIN plans p ON c.plan_id = p.id
WHERE p.price = (
    SELECT MIN(price)
    FROM plans
);


-- Q204. Find customers whose total successful payment is greater
-- than the average successful payment of all customers.
SELECT c.id, c.name, SUM(t.amount) AS total_paid
FROM customers c
JOIN transactions t ON c.id = t.customer_id
WHERE t.status = 'Success'
GROUP BY c.id, c.name
HAVING SUM(t.amount) > (
    SELECT AVG(customer_total)
    FROM (
        SELECT SUM(amount) AS customer_total
        FROM transactions
        WHERE status = 'Success'
        GROUP BY customer_id
    ) x
);


-- Q205. Find customers who have made at least one successful transaction
-- and are using a plan costing more than 1000.
SELECT DISTINCT c.id, c.name, p.name AS plan_name
FROM customers c
JOIN plans p ON c.plan_id = p.id
JOIN transactions t ON c.id = t.customer_id
WHERE t.status = 'Success'
AND p.price > 1000;

-- Q230. Find customers who used more than 80% of their plan data limit.
SELECT c.id, c.name,
       p.name AS plan_name,
       p.data_limit_gb,
       SUM(u.total_usage_gb) AS total_usage,
       ROUND(
           SUM(u.total_usage_gb) / p.data_limit_gb * 100,
           2
       ) AS usage_percentage

JOIN usage_logs u ON c.id = u.customer_id

GROUP BY c.id, c.name, p.name, p.data_limit_gb

-- Q283. Create procedure to find customers by plan.
CREATE PROCEDURE GetCustomersByPlan(IN plan_id_input INT)
BEGIN
    SELECT c.id,
           c.name,
           c.email,
           p.name AS plan_name
    FROM customers c
    JOIN plans p ON c.plan_id = p.id
    WHERE c.plan_id = plan_id_input;
END $$


-- Q284. Execute plan customer procedure.
CALL GetCustomersByPlan(1) $$


-- Q285. Create procedure to calculate total successful payment
-- for a customer.
CREATE PROCEDURE GetCustomerPayment(IN customer_id_input INT)
BEGIN
    SELECT c.id,
           c.name,
           COALESCE(SUM(t.amount), 0) AS total_paid
    FROM customers c
    LEFT JOIN transactions t
        ON c.id = t.customer_id
        AND t.status = 'Success'
    WHERE c.id = customer_id_input
    GROUP BY c.id, c.name;
END $$


-- Q286. Execute customer payment procedure.
CALL GetCustomerPayment(1) $$


-- Q287. Create procedure to find customers with usage above
-- a given value.
CREATE PROCEDURE GetHighUsageCustomers(IN usage_limit DECIMAL(10,2))
BEGIN
    SELECT c.id,
           c.name,
           SUM(u.total_usage_gb) AS total_usage
    FROM customers c
    JOIN usage_logs u ON c.id = u.customer_id
    GROUP BY c.id, c.name
    HAVING SUM(u.total_usage_gb) > usage_limit;
END $$


-- Q288. Execute high usage procedure.

-- Q289. Create procedure to count customers for a plan.
CREATE PROCEDURE CountCustomersByPlan(IN plan_id_input INT)
BEGIN
    SELECT p.id,
           p.name,
           COUNT(c.id) AS customer_count
    FROM plans p
    LEFT JOIN customers c ON p.id = c.plan_id
    WHERE p.id = plan_id_input
    GROUP BY p.id, p.name;
END $$


-- Q290. Execute count customer procedure.

-- Q291. Create a function to calculate GST-inclusive price.
CREATE FUNCTION CalculateGST(price_value DECIMAL(10,2))
RETURNS DECIMAL(10,2)
DETERMINISTIC
BEGIN
    RETURN ROUND(price_value * 1.10, 2);
END $$


-- Q292. Use the GST function.

       price,
       CalculateGST(price) AS price_with_gst
FROM plans $$


-- Q293. Create a function to calculate discount.
CREATE FUNCTION CalculateDiscount(
    price_value DECIMAL(10,2),
    discount_percent DECIMAL(5,2)
)
RETURNS DECIMAL(10,2)
DETERMINISTIC
BEGIN
    RETURN ROUND(
        price_value - (price_value * discount_percent / 100),
        2
    );
END $$


-- Q294. Apply 10% discount to all plans.
SELECT name,
       price,
       CalculateDiscount(price, 10) AS discounted_price
FROM plans $$


-- Q295. Calculate number of days between start date and due date.
CREATE FUNCTION SubscriptionDays(
    start_value DATE,
    due_value DATE
)
RETURNS INT
DETERMINISTIC
BEGIN
    RETURN DATEDIFF(due_value, start_value);
END $$


DELIMITER ;

FROM customers;

-- Q296. Create an audit table for transactions.
CREATE TABLE transaction_audit (
    id INT PRIMARY KEY AUTO_INCREMENT,
    transaction_id INT,
    customer_id INT,
    amount DECIMAL(10,2),
    action_type VARCHAR(20),
    action_time DATETIME
) $$


-- Q297. Create trigger to record new transactions.
CREATE TRIGGER after_transaction_insert
AFTER INSERT ON transactions
FOR EACH ROW
BEGIN
    INSERT INTO transaction_audit
    (
        transaction_id,
        customer_id,
        amount,
        action_type,
        action_time
    )
    VALUES
    (
        NEW.id,
        NEW.customer_id,
        NEW.amount,
        'INSERT',
        NOW()
    );
END $$

-- Q300. Create a trigger to record transaction updates.
CREATE TRIGGER after_transaction_update
AFTER UPDATE ON transactions
FOR EACH ROW
BEGIN
    INSERT INTO transaction_audit
    (
        transaction_id,
        customer_id,
        amount,
        action_type,
        action_time
    )
    VALUES
    (
        NEW.id,
        NEW.customer_id,
        NEW.amount,
        'UPDATE',
        NOW()
    );
END $$
